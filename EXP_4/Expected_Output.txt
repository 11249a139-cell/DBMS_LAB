(b) UNIQUE Violation

(c) PRIMARY KEY Violation
Error: Duplicate roll_no value '101'.

(d) CHECK Violation
Error: Age must be between 17 and 30.

(e) FOREIGN KEY Violation
Error: dept_id '9' does not exist in the Department table.

ALTER TABLE Student
